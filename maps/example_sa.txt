# Example schematic for sa_daemon --map maps/example_sa.txt
#
# All lengths are metres. The schematic is rotated by `yaw` and translated by
# `origin` into world coordinates. World +x is the device's initial forward
# axis and +y is to the left. This is a prior: sensor rays still refine the same
# occupancy cells, and walls never create person tracks.
#
# Commands:
#   origin x y
#   yaw degrees
#   scale schematic_units_to_metres
#   wall  x1 y1 x2 y2 [height_m]
#   door  x1 y1 x2 y2 [height_m]
#   room  Name x1 y1 x2 y2
#   label Name x y

origin 0 0
yaw 0
scale 1.0

# Outer shell, 10 m x 8 m.
wall -5 -4  5 -4
wall  5 -4  5  4
wall  5  4 -5  4
wall -5  4 -5 -4

# West partition with a 1 m doorway.
wall -1 -4 -1 -1
door -1 -1 -1  1
wall -1  1 -1  4

# East partition with a 1 m doorway.
wall 1 -4 1 1
wall 1  1 5 1
door 2.5 1 3.5 1

room Lab      -5 -4 -1 4
room Corridor -1 -4  1 4
room Office    1 -4  5 1
room Storage   1  1  5 4

label Exit   -0.85 0.0
label Ladder  3.7 -3.3
label Server  4.0  2.9
